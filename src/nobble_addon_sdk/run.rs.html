21/// Serve an addon on stdin and stdout, until told to stop.
22///
23/// Returns when the daemon sends `Shutdown` or closes the pipe. An addon's
24/// `main` is normally nothing but this.
25pub fn run<A: Addon>(addon: A) {
26    let stdin = BufReader::new(std::io::stdin());
27    let stdout = std::io::stdout();
28    // A broken pipe means the daemon went away, which is its business and not
29    // an error to shout about: exit quietly and let it restart us.
30    let _ = serve(addon, stdin, stdout);
31}

33/// [`run`], with the pipe supplied. For tests, and for anything hosting an
34/// addon over something other than stdio.
35///
36/// # Errors
37/// If the pipe fails. A malformed line is answered with a failure rather than
38/// returned — one bad frame should not end the conversation.
39pub fn serve<A: Addon, R: BufRead + Send + 'static, W: Write + Send + 'static>(
40    mut addon: A,
41    reader: R,
42    writer: W,
43) -> std::io::Result<()> {
44    let pipe = Arc::new(Mutex::new(Pipe {
45        reader,
46        writer,
47        line: String::new(),
48    }));
49    let credentials: CredentialHandle = Arc::new(WireCredentials {
50        pipe: Arc::clone(&pipe),
51    });
52
53    loop {
54        let Some(text) = read_line(&pipe)? else {
55            return Ok(()); // the daemon closed the pipe
56        };
57        let request: Request = match serde_json::from_str(&text) {
58            Ok(r) => r,
59            Err(e) => {
60                // Answered rather than fatal. A frame this end cannot parse is
61                // a bug in one message, and killing the conversation over it
62                // would turn a bad request into a dead addon.
63                send(
64                    &pipe,
65                    &Reply::Failed {
66                        kind: FailureKind::Failed,
67                        detail: format!("could not understand that: {e}"),
68                    },
69                )?;
70                continue;
71            }
72        };
73
74        let reply = match request {
75            // Version checking is the daemon's call, not ours: it is the one that
76            // can report a mismatch to a user and name the fix (FR-049). Answering
77            // honestly and letting it decide is the whole of our part.
78            Request::Hello { .. } => Reply::Welcome { version: PROTOCOL },
79            Request::Describe => Reply::Description(describe(&addon)),
80            Request::Availability => Reply::Availability(match addon.availability() {
81                Availability::Ready => AvailabilityDecl::Ready,
82                Availability::Unavailable(detail) => AvailabilityDecl::Unavailable { detail },
83            }),
84            Request::Applies { action } => Reply::Applies {
85                applies: addon.applies(&action),
86            },
87            Request::Status => Reply::Status {
88                status: addon.status(),
89            },
90            Request::Configure { values } => {
91                addon.configure(&values, Arc::clone(&credentials));
92                Reply::Done
93            }
94            Request::ReadSignals => Reply::Signals(match addon.read_signals() {
95                Reading::Values(values) => ReadingDecl::Values {
96                    values: values
97                        .into_iter()
98                        .map(|(id, on)| (id.to_owned(), on))
99                        .collect(),
100                },
101                Reading::Unavailable(detail) => ReadingDecl::Unavailable { detail },
102            }),
103            Request::Perform {
104                action,
105                params,
106                value,
107            } => perform(&mut addon, &action, &params, value),
108            Request::Shutdown => {
109                send(&pipe, &Reply::Done)?;
110                return Ok(());
111            }
112        };
113        send(&pipe, &reply)?;
114    }
115}

117fn perform<A: Addon>(
118    addon: &mut A,
119    action: &str,
120    params: &BTreeMap<String, crate::addon::ParamValue>,
121    value: Option<u16>,
122) -> Reply {
123    // ADR-0021's backward-compatibility half. A daemon that understands
124    // `device_actions` resolved this to a keystroke when the key was bound and
125    // never sends it here at all; a daemon built against SDK 1.0 dropped the
126    // field on the way in, because serde discards what it does not recognise,
127    // and will ask for a `perform` the author was told not to write.
128    //
129    // Answered here rather than left to fall through, because the fall-through
130    // is `AddonError::NoSuchAction` and that sentence is a lie: the action is
131    // right there in the list the user bound it from, and they would go looking
132    // for a missing action instead of an old Nobble.
133    //
134    // `Failed` rather than `Unavailable`: unavailable means try again later, and
135    // this never will be until the daemon is replaced.
136    if addon.device_actions().iter().any(|d| d.id == action) {
137        return Reply::Failed {
138            kind: FailureKind::Failed,
139            detail: format!(
140                "{action:?} is sent by the device, not by this addon. \
141                 This version of Nobble is older than the addon and cannot bind \
142                 it as a keystroke; update Nobble."
143            ),
144        };
145    }
146
147    let invocation = match value {
148        Some(v) => Invocation::moved(params, v),
149        None => Invocation::press(params),
150    };
151    match addon.perform(action, &invocation) {
152        Ok(()) => Reply::Done,
153        Err(e) => {
154            let (kind, detail) = match e {
155                AddonError::NoSuchAction(w) => (FailureKind::NoSuchAction, w),
156                AddonError::Unavailable(w) => (FailureKind::Unavailable, w),
157                AddonError::Failed(w) => (FailureKind::Failed, w),
158            };
159            Reply::Failed { kind, detail }
160        }
161    }
162}

164/// Derive the owned declaration from the borrowed one the author wrote.
165///
166/// This function is why `protocol`'s mirrors are not a second definition to
167/// keep in step: there is one authored source, and this is the only thing that
168/// produces the other.
169fn describe<A: Addon>(addon: &A) -> Description {
170    Description {
171        id: addon.id().to_owned(),
172        name: addon.name().to_owned(),
173        description: addon.description().to_owned(),
174        actions: addon.actions().iter().map(action_decl).collect(),
175        device_actions: addon
176            .device_actions()
177            .iter()
178            .map(device_action_decl)
179            .collect(),
180        choices: addon.choices().iter().map(choices_decl).collect(),
181        signals: addon.signals().iter().map(signal_decl).collect(),
182        settings: addon.settings().iter().map(setting_decl).collect(),
183        permissions: addon.permissions().iter().map(permission_decl).collect(),
184    }
185}

187fn permission_decl(p: &Permission) -> PermissionDecl {
188    match *p {
189        Permission::Network { host, reason } => PermissionDecl::Network {
190            host: host.to_owned(),
191            reason: reason.to_owned(),
192        },
193        Permission::Files {
194            path,
195            write,
196            reason,
197        } => PermissionDecl::Files {
198            path: path.to_owned(),
199            write,
200            reason: reason.to_owned(),
201        },
202        Permission::Launch { program, reason } => PermissionDecl::Launch {
203            program: program.to_owned(),
204            reason: reason.to_owned(),
205        },
206        Permission::Credentials { reason } => PermissionDecl::Credentials {
207            reason: reason.to_owned(),
208        },
209    }
210}

222/// The device-resolved half of the same one-way derivation.
223///
224/// A second function beside [`action_decl`] rather than a branch inside it,
225/// because the two produce different shapes — this one has no trigger and no
226/// parameters to convert. It is still the *only* thing that produces a
227/// [`DeviceActionDecl`], which is the property that matters: one authored
228/// source, one derivation per owned type, nothing hand-maintaining the mirror.
229fn device_action_decl(d: &DeviceAction) -> DeviceActionDecl {
230    DeviceActionDecl {
231        id: d.id.to_owned(),
232        name: d.name.to_owned(),
233        description: d.description.to_owned(),
234        keystroke: keystroke_decl(d.keystroke),
235        prerequisite: d.prerequisite.map(ToOwned::to_owned),
236    }
237}

239/// Exhaustive on purpose: [`KeystrokeDecl::Unknown`] is a decode state and has
240/// no arm here, so a new [`DeviceKeystroke`] variant fails to compile until
241/// somebody decides what crosses the pipe for it.
242fn keystroke_decl(k: DeviceKeystroke) -> KeystrokeDecl {
243    match k {
244        DeviceKeystroke::Tap {
245            key,
246            ctrl,
247            shift,
248            alt,
249            gui,
250        } => KeystrokeDecl::HidTap {
251            key,
252            ctrl,
253            shift,
254            alt,
255            gui,
256        },
257        DeviceKeystroke::Consumer { usage } => KeystrokeDecl::HidConsumer { usage },
258    }
259}

304/// The pipe, and the buffer reads reuse.
305struct Pipe<R, W> {
306    reader: R,
307    writer: W,
308    line: String,
309}
310
311fn read_line<R: BufRead, W: Write>(
312    pipe: &Arc<Mutex<Pipe<R, W>>>,
313) -> std::io::Result<Option<String>> {
314    let mut p = pipe
315        .lock()
316        .unwrap_or_else(std::sync::PoisonError::into_inner);
317    p.line.clear();
318    let Pipe { reader, line, .. } = &mut *p;
319    if reader.read_line(line)? == 0 {
320        return Ok(None);
321    }
322    Ok(Some(p.line.trim_end().to_owned()))
323}
324
325fn send<R: BufRead, W: Write, T: serde::Serialize>(
326    pipe: &Arc<Mutex<Pipe<R, W>>>,
327    message: &T,
328) -> std::io::Result<()> {
329    let mut p = pipe
330        .lock()
331        .unwrap_or_else(std::sync::PoisonError::into_inner);
332    let text = serde_json::to_string(message)?;
333    writeln!(p.writer, "{text}")?;
334    p.writer.flush()
335}

337/// [`Credentials`] over the pipe.
338///
339/// Every call is an [`Ask`] out and an [`Answer`] back, taken while the addon
340/// is part-way through a request. That works because the conversation is
341/// strictly nested — the daemon is waiting for our reply, so the next thing it
342/// sends is the answer to this — and it is what keeps `Credentials` a plain
343/// synchronous call in the addon's own code.
344struct WireCredentials<R, W> {
345    pipe: Arc<Mutex<Pipe<R, W>>>,
346}
347
348impl<R: BufRead + Send, W: Write + Send> WireCredentials<R, W> {
349    fn exchange(&self, ask: &Ask) -> Option<Answer> {
350        send(&self.pipe, &Reply::Ask(ask.clone())).ok()?;
351        let text = read_line(&self.pipe).ok()??;
352        serde_json::from_str(&text).ok()
353    }
354}
355
356impl<R: BufRead + Send, W: Write + Send> Credentials for WireCredentials<R, W> {
357    fn get(&self, key: &str) -> Option<String> {
358        match self.exchange(&Ask::Get {
359            key: key.to_owned(),
360        })? {
361            Answer::Value { value } => value,
362            // A refusal and an absence are the same to the caller: there is no
363            // credential to use. The daemon has already told the user why.
364            Answer::Stored | Answer::Refused { .. } => None,
365        }
366    }
367
368    fn set(&self, key: &str, value: &str) -> Result<(), String> {
369        match self.exchange(&Ask::Set {
370            key: key.to_owned(),
371            value: value.to_owned(),
372        }) {
373            Some(Answer::Stored) => Ok(()),
374            Some(Answer::Refused { detail }) => Err(detail),
375            Some(Answer::Value { .. }) | None => Err("the daemon did not answer".to_owned()),
376        }
377    }
378
379    fn clear(&self, key: &str) {
380        let _ = self.exchange(&Ask::Clear {
381            key: key.to_owned(),
382        });
383    }
384}

392    /// A writer the test can still read after `serve` has taken it.
393    #[derive(Clone, Default)]
394    struct Recorder(Arc<Mutex<Vec<u8>>>);
395
396    impl Write for Recorder {
397        fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
398            self.0
399                .lock()
400                .unwrap_or_else(std::sync::PoisonError::into_inner)
401                .extend_from_slice(buf);
402            Ok(buf.len())
403        }
404        fn flush(&mut self) -> std::io::Result<()> {
405            Ok(())
406        }
407    }
408
409    impl Recorder {
410        fn lines(&self) -> Vec<String> {
411            let bytes = self
412                .0
413                .lock()
414                .unwrap_or_else(std::sync::PoisonError::into_inner)
415                .clone();
416            String::from_utf8(bytes)
417                .expect("utf-8")
418                .lines()
419                .map(str::to_owned)
420                .collect()
421        }
422    }
