1//! Volume, through Windows Core Audio.
2//!
3//! # Why this and not Spotify's own volume
4//!
5//! Spotify's Web API can set Spotify's internal volume, and it needs an OAuth
6//! login, a network round trip, and **Premium** — it returns 403 on a free
7//! account. This needs none of them: it moves the slider Windows already keeps
8//! for every application, works offline, works on a free account, and works for
9//! anything that makes noise rather than only for Spotify.
10//!
11//! The two are not the same control and the interface should not pretend
12//! otherwise. This one does not follow playback to a phone or a Connect
13//! speaker; Spotify's own does. That is why the Spotify addon has its own
14//! volume action rather than this one growing a mode.
15//!
16//! # Why a fader and not two keys
17//!
18//! Volume is the reason [ADR-0012] exists. A pair of up/down keys can be
19//! device-resolved HID and needs no addon at all; what a fader gives is
20//! *absolute* position — the control ends up where your hand is, rather than
21//! wherever a count of key presses left it.
22//!
23//! That comes at a price the interface has to state: a fader bound here is
24//! host-resolved, so it stops working when the daemon does. A fader bound to
25//! MIDI keeps working with the client closed (ADR-0007). Both are legitimate
26//! and they are different promises.
27//!
28//! # The taper
29//!
30//! Windows' `ISimpleAudioVolume` is a **scalar**, and scalars are linear in
31//! amplitude while hearing is not. A fader mapped straight through spends its
32//! top half doing almost nothing audible and its bottom tenth doing everything,
33//! which feels broken long before anyone works out why.
34//!
35//! So the fraction is cubed. That is the same curve Windows' own volume mixer
36//! applies, and it is deliberately not a decibel taper: matching the slider a
37//! user already has beats being theoretically right about loudness.
38//!
39//! [ADR-0012]: ../../../docs/decisions/0012-addon-actions-carry-data.md
40

97/// Turn a fader fraction into an amplitude scalar.
98///
99/// Cubed, for the reason in the module documentation: a linear map spends the
100/// fader's top half doing nothing audible. Kept out of `platform` so it is
101/// testable without Windows, which matters because it is the one piece of this
102/// addon that is a *choice* rather than a call.
103#[must_use]
104pub fn taper(fraction: f32) -> f32 {
105    // Clamped rather than trusted. The fraction comes from a 14-bit position so
106    // it cannot exceed 1.0 today, but `SetMasterVolume` rejects out-of-range
107    // and would fail the whole action rather than the arithmetic.
108    let f = fraction.clamp(0.0, 1.0);
109    f * f * f
110}

112impl Addon for Volume {
113    fn id(&self) -> &'static str {
114        "volume"
115    }
116
117    fn name(&self) -> &'static str {
118        "Volume"
119    }
120
121    fn description(&self) -> &'static str {
122        "Moves the Windows volume — the system slider, or one application's. No account needed."
123    }
124
125    fn actions(&self) -> &'static [AddonAction] {
126        ACTIONS
127    }
128
129    fn availability(&self) -> Availability {
130        platform::availability()
131    }
132
133    fn perform(&mut self, action: &str, invocation: &Invocation<'_>) -> Result<(), AddonError> {
134        let app = invocation.param("app");
135        match action {
136            "set" => {
137                // The trigger check in `Registry::perform` guarantees a value
138                // for a continuous action, so this cannot be `None` in
139                // practice. Saying so rather than unwrapping keeps the addon
140                // honest if it is ever called directly.
141                let Some(fraction) = invocation.fraction() else {
142                    return Err(AddonError::Failed(
143                        "setting the volume needs a fader position".to_owned(),
144                    ));
145                };
146                platform::set(app, taper(fraction))
147            }
148            "mute" => platform::toggle_mute(app),
149            other => Err(AddonError::NoSuchAction(other.to_owned())),
150        }
151    }
152}

154#[cfg(windows)]
155mod platform {
156    use nobble_addon_sdk::{AddonError, Availability};
157    use windows::Win32::Media::Audio::Endpoints::IAudioEndpointVolume;
158    use windows::Win32::Media::Audio::{
159        DEVICE_STATE_ACTIVE, IAudioSessionControl2, IAudioSessionManager2, IMMDeviceEnumerator,
160        ISimpleAudioVolume, MMDeviceEnumerator, eMultimedia, eRender,
161    };
162    use windows::Win32::System::Com::{
163        CLSCTX_ALL, COINIT_MULTITHREADED, CoCreateInstance, CoInitializeEx, CoUninitialize,
164    };
165    use windows::core::Interface;
166
167    use nobble_addon_sdk::app_matches;
168
169    /// COM initialised for the duration of one call, and undone afterwards.
170    ///
171    /// The signal poll and the action worker are different threads, and COM
172    /// apartment state is per thread — so initialising once at startup would
173    /// leave whichever thread did not do it unable to make a call. Doing it per
174    /// call is a few microseconds and cannot get this wrong.
175    ///
176    /// `CoInitializeEx` returning `RPC_E_CHANGED_MODE` means the thread is
177    /// already in an apartment, which is fine: the call still works, and only
178    /// the matching `CoUninitialize` must be skipped.
179    struct Com(bool);
180
181    impl Com {
182        fn enter() -> Self {
183            // SAFETY: a plain COM initialisation with no pointer arguments. The
184            // matching CoUninitialize is in Drop, and is skipped when this call
185            // did not take ownership of the apartment.
186            let hr = unsafe { CoInitializeEx(None, COINIT_MULTITHREADED) };
187            Self(hr.is_ok())
188        }
189    }
190
191    impl Drop for Com {
192        fn drop(&mut self) {
193            if self.0 {
194                // SAFETY: paired with the successful CoInitializeEx above, on
195                // the same thread, exactly once.
196                unsafe { CoUninitialize() };
197            }
198        }
199    }
200
201    /// The default playback device's master volume.
202    fn endpoint() -> Option<IAudioEndpointVolume> {
203        // SAFETY: standard COM activation. Every pointer is produced by the
204        // call itself and dropped by the projection's reference counting.
205        unsafe {
206            let enumerator: IMMDeviceEnumerator =
207                CoCreateInstance(&MMDeviceEnumerator, None, CLSCTX_ALL).ok()?;
208            let device = enumerator
209                .GetDefaultAudioEndpoint(eRender, eMultimedia)
210                .ok()?;
211            device.Activate(CLSCTX_ALL, None).ok()
212        }
213    }
214
215    /// One application's audio session, found by name.
216    ///
217    /// Matched on the process's executable rather than on the session
218    /// identifier, because the session identifier is a device path with the
219    /// executable buried in it and the foreground watcher speaks in executables
220    /// (FR-014). The comparison is the SDK's `app_matches`, which is where it
221    /// belongs: `ParamKind::App` is an SDK concept, so how to compare one has
222    /// to be too, or every addon guesses separately and they disagree.
223    fn app_session(app: &str) -> Option<ISimpleAudioVolume> {
224        // SAFETY: standard COM activation and enumeration; every interface
225        // pointer comes from a call that succeeded and is reference counted by
226        // the projection.
227        unsafe {
228            let enumerator: IMMDeviceEnumerator =
229                CoCreateInstance(&MMDeviceEnumerator, None, CLSCTX_ALL).ok()?;
230            let device = enumerator
231                .GetDefaultAudioEndpoint(eRender, eMultimedia)
232                .ok()?;
233            let manager: IAudioSessionManager2 = device.Activate(CLSCTX_ALL, None).ok()?;
234            let sessions = manager.GetSessionEnumerator().ok()?;
235            for i in 0..sessions.GetCount().ok()? {
236                let Ok(control) = sessions.GetSession(i) else {
237                    continue;
238                };
239                let Ok(control2) = control.cast::<IAudioSessionControl2>() else {
240                    continue;
241                };
242                // `PWSTR::to_string` is fallible here because Windows hands
243                // back UTF-16 that need not be valid. A session whose name
244                // cannot be decoded is skipped rather than matched loosely:
245                // the alternative is aiming at whichever process happened to
246                // have an odd identifier.
247                let Ok(Ok(id)) = control2.GetSessionIdentifier().map(|s| s.to_string()) else {
248                    continue;
249                };
250                if app_matches(&id, app)
251                    && let Ok(volume) = control.cast::<ISimpleAudioVolume>()
252                {
253                    return Some(volume);
254                }
255            }
256            None
257        }
258    }

260    pub fn availability() -> Availability {
261        let _com = Com::enter();
262        match endpoint() {
263            Some(_) => Availability::Ready,
264            None => {
265                Availability::Unavailable("Windows is not reporting a playback device.".to_owned())
266            }
267        }
268    }

270    pub fn level(app: Option<&str>) -> Option<f32> {
271        let _com = Com::enter();
272        // SAFETY: reads through interface pointers that were just obtained.
273        unsafe {
274            match app {
275                Some(a) => app_session(a)?.GetMasterVolume().ok(),
276                None => endpoint()?.GetMasterVolumeLevelScalar().ok(),
277            }
278        }
279    }
280
281    /// The named application's session, or a failure that names it.
282    ///
283    /// Split out because both `set` and `toggle_mute` want exactly this, and
284    /// the message matters: "spotify is not playing audio" sends someone to
285    /// start playback, while a generic failure sends them to the binding.
286    fn require_session(app: &str) -> Result<ISimpleAudioVolume, AddonError> {
287        app_session(app)
288            .ok_or_else(|| AddonError::Unavailable(format!("{app} is not playing audio")))
289    }
290
291    /// The default endpoint, or a failure.
292    fn require_endpoint() -> Result<IAudioEndpointVolume, AddonError> {
293        endpoint().ok_or_else(|| {
294            AddonError::Unavailable("Windows is not reporting a playback device".to_owned())
295        })
296    }
297
298    pub fn set(app: Option<&str>, scalar: f32) -> Result<(), AddonError> {
299        let _com = Com::enter();
300        if let Some(a) = app {
301            let session = require_session(a)?;
302            // SAFETY: a scalar set on an interface just obtained. The null GUID
303            // means "no event context", which is correct: nothing here is
304            // listening for its own change.
305            return unsafe { session.SetMasterVolume(scalar, std::ptr::null()) }
306                .map_err(|e| AddonError::Failed(e.message()));
307        }
308        let endpoint = require_endpoint()?;
309        // SAFETY: as above.
310        unsafe { endpoint.SetMasterVolumeLevelScalar(scalar, std::ptr::null()) }
311            .map_err(|e| AddonError::Failed(e.message()))
312    }
313
314    pub fn toggle_mute(app: Option<&str>) -> Result<(), AddonError> {
315        let _com = Com::enter();
316        if let Some(a) = app {
317            let session = require_session(a)?;
318            // SAFETY: read-then-write through one interface pointer, on one
319            // thread, with no intervening call that could invalidate it.
320            return unsafe {
321                let muted = session
322                    .GetMute()
323                    .map_err(|e| AddonError::Failed(e.message()))?;
324                session
325                    .SetMute(!muted.as_bool(), std::ptr::null())
326                    .map_err(|e| AddonError::Failed(e.message()))
327            };
328        }
329        let endpoint = require_endpoint()?;
330        // SAFETY: as above.
331        unsafe {
332            let muted = endpoint
333                .GetMute()
334                .map_err(|e| AddonError::Failed(e.message()))?;
335            endpoint
336                .SetMute(!muted.as_bool(), std::ptr::null())
337                .map_err(|e| AddonError::Failed(e.message()))
338        }
339    }
340
341    // Silences the unused-import warning on the enumeration constant, which is
342    // referenced only for documentation of what `GetSessionEnumerator` returns.
343    const _: u32 = DEVICE_STATE_ACTIVE.0;
344}

369#[cfg(test)]
370mod tests {
371    use super::*;
372
373    #[test]
374    fn the_fader_ends_are_silence_and_full() {
375        // Whatever the curve does in between, both ends have to be exact. A
376        // taper that bottomed out at 0.001 would leave a fader pulled all the
377        // way down still audible, which reads as broken hardware.
378        assert!((taper(0.0) - 0.0).abs() < f32::EPSILON);
379        assert!((taper(1.0) - 1.0).abs() < f32::EPSILON);
380    }
381
382    #[test]
383    fn the_taper_is_not_linear() {
384        // The whole point. Halfway up the fader is a quarter of the amplitude,
385        // which is roughly where a person expects "half volume" to sit --
386        // linear would put it at 0.5 and sound almost as loud as the top.
387        let half = taper(0.5);
388        assert!(half < 0.2, "halfway should be quiet, got {half}");
389        assert!(half > 0.1, "and not silent, got {half}");
390    }
391
392    #[test]
393    fn the_taper_never_goes_backwards() {
394        // A fader that got quieter as it was pushed up would be the funniest
395        // possible bug and the hardest to believe from a report.
396        let mut last = -1.0;
397        for step in 0..=100 {
398            #[allow(clippy::cast_precision_loss)]
399            let now = taper(step as f32 / 100.0);
400            assert!(now >= last, "went backwards at {step}");
401            last = now;
402        }
403    }
404
405    #[test]
406    fn an_impossible_fraction_is_clamped_rather_than_passed_on() {
407        // `SetMasterVolume` rejects out of range, so an unclamped value would
408        // fail the whole action rather than the arithmetic -- and the user
409        // would see "the fader does nothing" with no clue why.
410        assert!((taper(1.5) - 1.0).abs() < f32::EPSILON);
411        assert!((taper(-0.5) - 0.0).abs() < f32::EPSILON);
412    }
413
414    #[test]
415    fn setting_the_volume_wants_a_fader_and_muting_wants_a_key() {
416        // The distinction ADR-0012 introduced, on the addon it was introduced
417        // for. Getting these the wrong way round would let someone bind mute to
418        // a fader, where a sweep would toggle it a hundred times.
419        let set = ACTIONS.iter().find(|a| a.id == "set").unwrap();
420        let mute = ACTIONS.iter().find(|a| a.id == "mute").unwrap();
421        assert_eq!(set.trigger, Trigger::Continuous);
422        assert_eq!(mute.trigger, Trigger::Momentary);
423    }
424
425    #[test]
426    fn the_target_is_optional_so_the_system_slider_is_the_default() {
427        for a in ACTIONS {
428            let target = a.param("app").expect("every action can be aimed");
429            assert!(!target.required);
430            assert_eq!(target.kind, ParamKind::App);
431        }
432    }

434    #[test]
435    fn an_unknown_action_is_refused() {
436        let mut addon = Volume::new();
437        assert!(matches!(
438            addon.perform("teleport", &Invocation::bare()),
439            Err(AddonError::NoSuchAction(_))
440        ));
