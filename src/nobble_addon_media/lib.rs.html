1//! Transport control through the Windows media session.
2//!
3//! # Why this and not the Spotify Web API
4//!
5//! Spotify's Web API can do far more — playlists, saving a track, moving
6//! playback between devices — and it needs a registered application, an OAuth
7//! login per user, and a refresh token the daemon has to keep. None of that
8//! works while the network is down, and all of it fails in ways a user has to
9//! understand.
10//!
11//! `GlobalSystemMediaTransportControlsSessionManager` is what Windows itself
12//! uses for the media overlay. It needs no account, no client id and no
13//! network, it reports the track that is actually playing, and it works for
14//! whatever is playing rather than only Spotify.
15//!
16//! It is deliberately the *first* addon and not the only one: a Web API addon
17//! is worth having for what this cannot reach, and the two can coexist because
18//! a binding names the addon it wants.
19//!
20//! # Why not just bind a media key
21//!
22//! For play/pause alone, `hid_consumer` is better — the device sends it, so it
23//! works with Nobble closed (ADR-0007), and this addon does not. What the media
24//! session adds is *knowing*: which application is playing, and what the track
25//! is called. That is what a key with a display is for, and a keyboard cannot
26//! find it out alone.
27

33/// Which application a key controls.
34///
35/// Optional, and absent is the useful default: "whatever is playing" is what
36/// most people want from a play/pause key most of the time, and it is the
37/// behaviour this addon had before parameters existed.
38///
39/// Naming one turns the key into a control for *that* application, which keeps
40/// working when a browser tab starts a video and takes the current session
41/// away. Before this, a "skip track" key would skip the video instead — a key
42/// that does the wrong thing rather than nothing, which is worse.
43const TARGET: &[AddonParam] = &[AddonParam {
44    id: "app",
45    name: "Application",
46    description: "Which application to control. Leave empty for whatever is playing.",
47    kind: ParamKind::App,
48    required: false,
49    ..AddonParam::BASE
50}];
51

55        name: "Play / pause",
56        description: "Toggles playback.",
57        trigger: Trigger::Momentary,
58        params: TARGET,

89/// What is playing, if anything.
90#[derive(Debug, Clone, PartialEq, Eq, Default)]
91pub struct NowPlaying {
92    /// The application that owns the session, e.g. `Spotify.exe`.
93    pub source: String,
94    /// Track title, empty if the session does not report one.
95    pub title: String,
96    /// Artist, empty if unknown.
97    pub artist: String,
98    /// Whether it is playing rather than paused.
99    pub playing: bool,
100}

102/// Media transport control, through whatever Windows says is playing.
103#[derive(Debug, Default)]
104pub struct MediaSession {
105    _private: (),
106}
107
108impl MediaSession {
109    /// A new addon. Does not touch the platform until asked.
110    #[must_use]
111    pub fn new() -> Self {
112        Self::default()
113    }
114
115    /// What is playing right now, if anything.
116    #[must_use]
117    pub fn now_playing(&self) -> Option<NowPlaying> {
118        platform::now_playing(None)
119    }
120
121    /// What a named application is playing, if it is.
122    ///
123    /// `app` is an application identity as the foreground watcher reports it,
124    /// such as `exe:spotify.exe`.
125    #[must_use]
126    pub fn now_playing_for(&self, app: &str) -> Option<NowPlaying> {
127        platform::now_playing(Some(app))
128    }
129}

131impl Addon for MediaSession {
132    fn id(&self) -> &'static str {
133        "media"
134    }
135
136    fn name(&self) -> &'static str {
137        "Media"
138    }
139
140    fn description(&self) -> &'static str {
141        "Controls whatever is playing — Spotify, a browser tab, anything that registers with Windows. No account needed."
142    }
143
144    fn actions(&self) -> &'static [AddonAction] {
145        ACTIONS
146    }
147
148    fn signals(&self) -> &'static [AddonSignal] {
149        SIGNALS
150    }
151
152    fn availability(&self) -> Availability {
153        platform::availability()
154    }
155
156    fn perform(&mut self, action: &str, invocation: &Invocation<'_>) -> Result<(), AddonError> {
157        if !ACTIONS.iter().any(|a| a.id == action) {
158            return Err(AddonError::NoSuchAction(action.to_owned()));
159        }
160        platform::perform(action, invocation.param("app"))
161    }
162
163    fn read_signals(&mut self) -> Reading {
164        // `playing`, not `now_playing`. This runs twice a second and needs one
165        // boolean; the version that fetched the track title and artist and
166        // then discarded them is what SC-005 caught.
167        match platform::playing(None) {
168            Some(playing) => Reading::Values(vec![("playing", playing)]),
169            None => Reading::Unavailable("nothing is playing".to_owned()),
170        }
171    }
172}

174#[cfg(windows)]
175mod platform {
176    use nobble_addon_sdk::{AddonError, Availability};
177    // `join()` blocks until the WinRT operation completes. It runs on the
178    // host-action worker thread, never the input path — FR-022 exists so a
179    // slow action cannot delay the next keypress.
180    use windows::Media::Control::GlobalSystemMediaTransportControlsSession as Session;
181    use windows::Media::Control::GlobalSystemMediaTransportControlsSessionManager as Manager;
182    use windows::Media::Control::GlobalSystemMediaTransportControlsSessionPlaybackStatus as Status;
183
184    use std::cell::RefCell;
185    use std::time::{Duration, Instant};
186
187    use super::NowPlaying;
188    use nobble_addon_sdk::app_matches;
189
190    /// How long a requested session manager is kept before being thrown away.
191    ///
192    /// **`RequestAsync` leaks**, and this is the number that bounds it.
193    /// Measured under SC-005: calling it twice a second cost +0.313 points of
194    /// CPU and +0.72 MB of working set **per minute**, linearly, for as long as
195    /// the measurement ran — 43 MB an hour, which reaches the addon memory cap
196    /// in under six hours. With the manager kept, the same window is flat at
197    /// 0.08% of a core and 15.0 MB.
198    ///
199    /// Five minutes rather than "for ever" because of what the previous comment
200    /// here was worried about — see [`manager`]. At this rate the call happens
201    /// twelve times an hour instead of seven thousand two hundred.
202    const MANAGER_TTL: Duration = Duration::from_secs(300);
203
204    thread_local! {
205        /// The manager, and when it was requested.
206        static MANAGER: RefCell<Option<(Manager, Instant)>> = const { RefCell::new(None) };
207    }
208
209    /// The session manager, requested at most once every [`MANAGER_TTL`].
210    ///
211    /// # What the comment that used to be here got right, and what it conflated
212    ///
213    /// It said a cached object "goes stale when the playing application
214    /// changes, and the failure is silent". That is true of a **session** and
215    /// not of the **manager**: a session belongs to one application and dies
216    /// with it, whereas the manager is the per-user service that is *asked*
217    /// which sessions exist. Nothing below caches a session — every call still
218    /// goes through `GetCurrentSession` or `GetSessions` on a live manager, so
219    /// the freshness that comment was protecting is exactly preserved.
220    ///
221    /// The TTL is there because the distinction is an argument rather than a
222    /// proof. If the manager can go bad in some way this does not anticipate,
223    /// it self-heals within five minutes instead of lasting until the addon is
224    /// restarted — and a failed call cannot be used as the trigger, because
225    /// "nothing is playing" arrives as an error too, so re-requesting on error
226    /// would re-request on every idle poll and put the leak straight back.
227    fn manager() -> Option<Manager> {
228        MANAGER.with(|slot| {
229            let mut slot = slot.borrow_mut();
230            if let Some((manager, at)) = slot.as_ref()
231                && at.elapsed() < MANAGER_TTL
232            {
233                return Some(manager.clone());
234            }
235            let fresh = Manager::RequestAsync().ok()?.join().ok()?;
236            *slot = Some((fresh.clone(), Instant::now()));
237            Some(fresh)
238        })
239    }
240
241    /// The session a binding meant, or nothing.
242    ///
243    /// With no target this is `GetCurrentSession` — whatever last made noise,
244    /// which is the right default for a bare play/pause key. With one it walks
245    /// every session and picks the match, which is what makes a key keep
246    /// controlling Spotify while a browser tab plays over the top.
247    ///
248    /// Asked afresh every call. Only the manager is kept; see [`manager`].
249    fn session(target: Option<&str>) -> Option<Session> {
250        let manager = manager()?;
251        let Some(app) = target else {
252            return manager.GetCurrentSession().ok();
253        };
254        manager.GetSessions().ok()?.into_iter().find(|s| {
255            s.SourceAppUserModelId()
256                .is_ok_and(|id| app_matches(&id.to_string(), app))
257        })
258    }
259
260    /// Whether something is playing, without asking what.
261    ///
262    /// The polled path, and it exists because the one it replaced fetched the
263    /// track's title and artist through `TryGetMediaPropertiesAsync` and then
264    /// used neither. Twice a second, for a boolean.
265    ///
266    /// It is also slightly *more* truthful than what it replaced: a session
267    /// whose metadata cannot be read used to report "nothing is playing", when
268    /// what is actually known is that something is playing and is not saying
269    /// what.
270    pub fn playing(target: Option<&str>) -> Option<bool> {
271        let session = session(target)?;
272        session
273            .GetPlaybackInfo()
274            .ok()?
275            .PlaybackStatus()
276            .ok()
277            .map(|s| s == Status::Playing)
278    }
279
280    pub fn availability() -> Availability {
281        match session(None) {
282            Some(_) => Availability::Ready,
283            None => Availability::Unavailable(
284                "Nothing is playing. Start Spotify, or anything else that reports to Windows."
285                    .to_owned(),
286            ),
287        }
288    }
289
290    pub fn now_playing(target: Option<&str>) -> Option<NowPlaying> {
291        let session = session(target)?;
292        let properties = session.TryGetMediaPropertiesAsync().ok()?.join().ok()?;
293        let playing = session
294            .GetPlaybackInfo()
295            .ok()
296            .and_then(|i| i.PlaybackStatus().ok())
297            .is_some_and(|s| s == Status::Playing);
298        Some(NowPlaying {
299            source: session
300                .SourceAppUserModelId()
301                .map(|s| s.to_string())
302                .unwrap_or_default(),
303            title: properties
304                .Title()
305                .map(|s| s.to_string())
306                .unwrap_or_default(),
307            artist: properties
308                .Artist()
309                .map(|s| s.to_string())
310                .unwrap_or_default(),
311            playing,
312        })
313    }
314
315    pub fn perform(action: &str, target: Option<&str>) -> Result<(), AddonError> {
316        let Some(session) = session(target) else {
317            // Names the application when one was asked for. "Nothing is
318            // playing" is a confusing thing to read when Spotify is plainly
319            // open and the key was bound to Chrome.
320            return Err(AddonError::Unavailable(match target {
321                Some(app) => format!("{app} is not playing anything Nobble can control"),
322                None => "nothing is playing for Nobble to control".to_owned(),
323            }));
324        };
325        // Each of these returns whether the session accepted it. A `false` is
326        // not an error from the API's point of view but it is one from the
327        // user's: the key did nothing.
328        let accepted = match action {
329            "play_pause" => session.TryTogglePlayPauseAsync().and_then(|op| op.join()),
330            "next" => session.TrySkipNextAsync().and_then(|op| op.join()),
331            "previous" => session.TrySkipPreviousAsync().and_then(|op| op.join()),
332            "stop" => session.TryStopAsync().and_then(|op| op.join()),
333            other => return Err(AddonError::NoSuchAction(other.to_owned())),
334        };
335        match accepted {
336            Ok(true) => Ok(()),
337            Ok(false) => Err(AddonError::Failed(format!(
338                "the application playing refused {action}"
339            ))),
340            Err(e) => Err(AddonError::Failed(e.message())),
341        }
342    }
343}

370#[cfg(test)]
371mod tests {
372    use super::*;

374    #[test]
375    fn every_action_has_a_stable_id_and_something_to_read() {
376        // Ids are stored in bindings, so a rename breaks someone's key
377        // silently. Names and descriptions are what a menu shows; an empty one
378        // is a blank row.
379        for a in ACTIONS {
380            assert!(!a.id.is_empty());
381            assert!(!a.name.is_empty());
382            assert!(!a.description.is_empty());
383            assert!(
384                a.id.chars().all(|c| c.is_ascii_lowercase() || c == '_'),
385                "{:?} should be a stable snake_case id",
386                a.id
387            );
388        }
389    }
390
391    #[test]
392    fn ids_are_unique() {
393        let mut seen: Vec<&str> = Vec::new();
394        for a in ACTIONS {
395            assert!(!seen.contains(&a.id), "duplicate action id {:?}", a.id);
396            seen.push(a.id);
397        }
398    }
399
400    #[test]
401    fn an_unknown_action_is_refused_before_the_platform_is_touched() {
402        // Otherwise a binding saved against a newer version of this addon
403        // reaches the media session as a typo and fails with whatever the
404        // platform says, rather than with what is actually wrong.
405        let mut addon = MediaSession::new();
406        match addon.perform("teleport", &Invocation::bare()) {
407            Err(AddonError::NoSuchAction(a)) => assert_eq!(a, "teleport"),
408            other => panic!("expected NoSuchAction, got {other:?}"),
409        }
410    }
411
412    #[test]
413    fn every_action_takes_an_optional_target() {
414        // Optional is the point. "Whatever is playing" is what most people want
415        // from a play/pause key, and making the target required would break
416        // every binding written before it existed.
417        for a in ACTIONS {
418            let target = a.param("app").expect("every action can be aimed");
419            assert!(!target.required);
420            assert_eq!(target.kind, ParamKind::App);
421            assert_eq!(a.trigger, Trigger::Momentary);
422        }
423    }
